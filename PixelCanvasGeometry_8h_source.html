127[[nodiscard]] constexpr double pixelCanvasScale(const PixelCanvasView& view) noexcept {
128 if (view.zoom <= 0 || view.zoomDenominator <= 0) {
129 return 0.0;
130 }
131 return static_cast<double>(view.zoom) / static_cast<double>(view.zoomDenominator);
132}

142[[nodiscard]] constexpr PixelCanvasView pixelCanvasZoomIn(const PixelCanvasView& view) noexcept {
143 PixelCanvasView next = view;
144 if (next.zoomDenominator > 1) {
145 next.zoomDenominator /= 2;
146 return next;
147 }
148 next.zoom = clampPixelCanvasZoom(next.zoom + 1);
149 return next;
150}
151
157[[nodiscard]] constexpr PixelCanvasView pixelCanvasZoomOut(const PixelCanvasView& view) noexcept {
158 PixelCanvasView next = view;
159 if (next.zoom > PIXEL_CANVAS_MIN_ZOOM) {
160 next.zoom = clampPixelCanvasZoom(next.zoom - 1);
161 return next;
162 }
163 if (next.zoomDenominator < PIXEL_CANVAS_MAX_DENOMINATOR) {
164 next.zoomDenominator *= 2;
165 }
166 return next;
167}
168
171[[nodiscard]] constexpr bool pixelCanvasGridVisible(const PixelCanvasView& view) noexcept {
172 return view.zoomDenominator == 1 && view.zoom >= PIXEL_CANVAS_GRID_VISIBLE_MIN_ZOOM;
