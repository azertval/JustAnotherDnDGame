129 [[nodiscard]] int skinIndexOf(std::string_view asset, core::TileType type) const noexcept {
130 const bool masked = hasSilhouette(type);
131 for (std::size_t index = 0; index < skins.size(); ++index) {
132 const SkinTexture& skin = skins[index];
133 if (skin.asset != asset) {
134 continue;
135 }
136 if (masked ? (skin.maskType == type) : !skin.maskType.has_value()) {
137 return static_cast<int>(index);
138 }
139 }
140 return -1;
141 }

151 [[nodiscard]] int objectIndexOf(std::string_view asset) const noexcept {
152 for (std::size_t index = 0; index < objects.size(); ++index) {
153 if (objects[index].asset == asset) {
154 return static_cast<int>(index);
155 }
156 }
157 return -1;
158 }

161 [[nodiscard]] TextureHandle textureFor(const TileAppearance& appearance) const noexcept {
162 switch (appearance.source) {
163 case AppearanceSource::Atlas:
164 return atlas;
165 case AppearanceSource::MissingTexture:
166 return missing;
167 case AppearanceSource::Skin:
168 return skinAt(appearance.skinIndex) ? skinAt(appearance.skinIndex)->texture
169 : missing;
170 case AppearanceSource::Override:
171 return objectAt(appearance.skinIndex) ? objectAt(appearance.skinIndex)->texture
172 : missing;
173 }
174 return missing;
175 }

212 // Acces borne au tableau de skins : un index invalide rend nullptr, ce qui fait retomber les
213 // accesseurs ci-dessus sur le damier plutot que de lire hors des bornes.
214 [[nodiscard]] const SkinTexture* skinAt(int index) const noexcept {
215 if (index < 0 || static_cast<std::size_t>(index) >= skins.size()) {
216 return nullptr;
217 }
218 return &skins[static_cast<std::size_t>(index)];
219 }
