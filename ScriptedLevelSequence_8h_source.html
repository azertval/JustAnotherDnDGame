38// Script constant : avancer à droite, rien d'autre (déplacement, pente, arrondi, interrupteur —
39// la trajectoire seule suffit, aucun timing particulier).
40ReactiveInput rightOnly() {
41 return [](int, const core::Player&, float, float) { return core::PlayerInput{1.0f}; };
42}

44// Vrai quand le personnage approche le bord droit @p edge (en cases) d'une plateforme, dans une
45// fenetre de @p window cases. Declencher un saut ou une ruee AVANT cette fenetre gaspille la portee
46// horizontale sur du sol encore solide -- constat deja fait au LOT-65 TACHE-02 sur demo-budget, et
47// qui vaut pour toute fosse.
48bool atLedge(float x, float edge, float window) {
49 return x >= edge - window && x <= edge - 0.05f;
50}
51
52// Un danger temporise est-il mortel au pas @p step ? Meme formule que
53// `core::DangerController::isBlinkActive`. Un tableau qui pose un danger temporise SUR le chemin
54// exige d'attendre sa fenetre inoffensive, ce qu'aucune entree reactive ne peut deviner depuis la
55// seule position du personnage : le script doit pouvoir le calculer.
56bool blinkActive(int step, int phase, int period, int activeDuration) {
57 const int cursor = (((step - phase) % period) + period) % period;
58 return cursor < activeDuration;
59}
60
61// Distance parcourue par une plateforme mobile le long de son segment au pas @p step, meme formule
62// que `core::PlatformController::boxAtStep` (aller-retour triangulaire 0 -> distance -> 0). Meme
63// raison que ci-dessus : attendre qu'une plateforme revienne ne se deduit pas de la position du
64// personnage.
65float platformOffset(int step, float speed, float distance) {
66 const float travelled = static_cast<float>(step) * speed / 60.0f;
67 const float cycle = distance * 2.0f;
68 float phase = std::fmod(travelled, cycle);
69 if (phase < 0.0f) {
70 phase += cycle;
71 }
72 return phase <= distance ? phase : cycle - phase;
73}

353 // 19. Dangers avances (EX-GP-051/052/053) : l'interrupteur ARME les dangers commutes places

366 // Attendre devant chaque danger temporise que sa fenetre mortelle soit passee, avec
367 // assez de marge pour traverser la case avant qu'elle ne revienne.
368 constexpr int PERIOD = 180;
369 constexpr int ACTIVE = 45;
370 constexpr int CROSSING = 40; // pas necessaires pour degager la case
371 const int phases[3] = {0, 60, 120};
372 const float holds[3] = {13.4f, 19.4f, 25.4f};
373 for (int index = 0; index < 3; ++index) {
374 if (x < holds[index] || x > holds[index] + 0.2f) {
375 continue;
376 }
377 if (blinkActive(step, phases[index], PERIOD, ACTIVE) ||
378 blinkActive(step + CROSSING, phases[index], PERIOD, ACTIVE)) {
379 in.moveX = 0.0f; // patienter : la case est (ou redevient) mortelle
380 }
381 }
382 return in;

888 // Double saut depuis le bloc (2, 18) : il traverse les portes ouvertes (2-4, 16)
889 // et débouche à la rangée 15. La dérive vers la droite suffit alors à toucher la
890 // sortie (5, 15).

892 // Le bloc (2, 18) ne fait qu'une case : marcher à droite en tombe. On saute
893 // d'abord à la verticale (les portes ouvertes 2-4 laissent passer), puis, une

899 in.moveX = y <= 15.2f ? 1.0f : 0.0f;
900 in.dashPressed = y <= 15.2f && x < 4.6f;
901 return in;
902 }
