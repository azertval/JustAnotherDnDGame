# C++ types consumed by the application QML. This module is deliberately separate from Jadg.Ui:
# Jadg.Ui is a reusable, design-only module; Jadg.Runtime is application/runtime-only.

set(JADG_RUNTIME_SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/../../Game/GameViewportItem.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Game/GameViewportItem.cpp
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/CharacterSheetModel.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/CharacterSheetModel.cpp
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/InventoryModel.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/InventoryModel.cpp
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/OptionsModel.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/OptionsModel.cpp
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/PendingData.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/PendingData.cpp
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/ScreenRouter.h
    ${CMAKE_CURRENT_LIST_DIR}/../../Presentation/ScreenRouter.cpp
)

qt_add_qml_module(JadgRuntime
    URI Jadg.Runtime
    VERSION 1.0
    RESOURCE_PREFIX "/qt/qml"
    NO_PLUGIN
    SOURCES ${JADG_RUNTIME_SOURCES}
)

target_include_directories(JadgRuntime PUBLIC ${PROJECT_SOURCE_DIR}/Source)
target_link_libraries(JadgRuntime
    PRIVATE
        HmiLib
        Core
        Qt6::Qml
        Qt6::Quick
        Qt6::GuiPrivate
        project_warnings
        project_options
)

# qmltyperegistrar metadata is generated by qt_add_qml_module. Keep the expected type-info path
# visible in the configure output so a missing registration artifact cannot hide behind a successful
# C++ build.
qt_query_qml_module(JadgRuntime TYPEINFO JADG_RUNTIME_QMLTYPES QMLDIR JADG_RUNTIME_QMLDIR)
message(STATUS "Jadg.Runtime qmltypes: ${JADG_RUNTIME_QMLTYPES}")
message(STATUS "Jadg.Runtime qmldir:   ${JADG_RUNTIME_QMLDIR}")

# Fail the build if the generated type metadata is unexpectedly absent. The check is attached to
# the module target, so it runs only after qmltyperegistrar has had a chance to generate the file.
add_custom_command(TARGET JadgRuntime POST_BUILD
    COMMAND ${CMAKE_COMMAND}
        -DEXPECTED_QMLTYPES="${JADG_RUNTIME_QMLTYPES}"
        -P "${CMAKE_CURRENT_LIST_DIR}/VerifyQmlTypes.cmake"
    VERBATIM
)

# The application links the backing target directly. There is no plugin-only dependency and no
# second compilation of the runtime sources in the application target.
