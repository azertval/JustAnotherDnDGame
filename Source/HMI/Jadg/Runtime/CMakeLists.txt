# C++ types consumed by the application QML. This module is deliberately separate from Jadg.Ui:
# Jadg.Ui is a reusable, design-only module; Jadg.Runtime is application/runtime-only.
#
# Keep the backing target static. The game links the module directly, which makes the generated
# QML registration deterministic and avoids a runtime DLL plugin dependency for the game.
add_library(JadgRuntime STATIC)

set(JADG_RUNTIME_SOURCES
    ../../Game/GameViewportItem.h
    ../../Game/GameViewportItem.cpp
    ../../Presentation/CharacterSheetModel.h
    ../../Presentation/CharacterSheetModel.cpp
    ../../Presentation/InventoryModel.h
    ../../Presentation/InventoryModel.cpp
    ../../Presentation/OptionsModel.h
    ../../Presentation/OptionsModel.cpp
    ../../Presentation/PendingData.h
    ../../Presentation/PendingData.cpp
    ../../Presentation/ScreenRouter.h
    ../../Presentation/ScreenRouter.cpp
)

qt_add_qml_module(JadgRuntime
    URI Jadg.Runtime
    VERSION 1.0
    CLASS_NAME JadgRuntimePlugin
    DEPENDENCIES
        QtQml
        QtQuick
    SOURCES ${JADG_RUNTIME_SOURCES}
)

target_include_directories(JadgRuntime PUBLIC ${PROJECT_SOURCE_DIR}/Source)

# qmltyperegistrar emits bare-filename includes (e.g. `#include <GameViewportItem.h>`) for every
# QML_ELEMENT/QML_NAMED_ELEMENT header listed in SOURCES, regardless of the relative path used to
# reference them above. Without these directories on the include path, __has_include() silently
# fails to find the headers and jadgruntime_qmltyperegistrations.cpp compiles with every registered
# type left undeclared. Keep PRIVATE: only this target's generated registration file needs it.
target_include_directories(JadgRuntime PRIVATE
    ${PROJECT_SOURCE_DIR}/Source/HMI/Game
    ${PROJECT_SOURCE_DIR}/Source/HMI/Presentation
)
target_link_libraries(JadgRuntime
    PRIVATE
        HmiLib
        Core
        Qt6::Qml
        Qt6::Quick
        Qt6::GuiPrivate
        project_warnings
        project_options
)

# qmltyperegistrar metadata is generated by qt_add_qml_module. Keep the expected type-info path
# visible in the configure output so a missing registration artifact cannot hide behind a successful
# C++ build.
qt_query_qml_module(JadgRuntime TYPEINFO JADG_RUNTIME_QMLTYPES QMLDIR JADG_RUNTIME_QMLDIR)
message(STATUS "Jadg.Runtime qmltypes: ${JADG_RUNTIME_QMLTYPES}")
message(STATUS "Jadg.Runtime qmldir:   ${JADG_RUNTIME_QMLDIR}")

# Fail the build if the generated type metadata is unexpectedly absent. The check is attached to
# the module target, so it runs only after qmltyperegistrar has had a chance to generate the file.
add_custom_command(TARGET JadgRuntime POST_BUILD
    COMMAND ${CMAKE_COMMAND}
        -DEXPECTED_QMLTYPES="${JADG_RUNTIME_QMLTYPES}"
        -P "${CMAKE_CURRENT_LIST_DIR}/VerifyQmlTypes.cmake"
    VERBATIM
)
